def test_no_closed_notice_never_calls_actual_program(case):
    with pytest.raises(ValueError):
        b.build(case.root, case.manifest)
    assert not case.calls


def test_six_mixed_dispositions_and_portable_closure(case):
    x = b.build(case.root, case.manifest, root_confirmed_six_closed=True)
    r = x["review"]
    assert len(r["accepted_records"]) == 6 and len(r["attempt_records"]) == 4
    assert r["calls_charged"] == r["calls_started"] == 6
    assert (
        r["original_G0_strict_review"] == T.BASELINE
        and r["original_G2_stable_review"] == T.G2_REVIEW
    )
    assert all(
        p.startswith(b.EVIDENCE + "/") and r["files"][p] == v["sha256"]
        for p, v in x["copy_plan"].items()
    )
    assert not (case.root / T.G_REVIEW).exists() and not (case.root / b.EVIDENCE).exists()


@pytest.mark.parametrize(
    "fault",
    [
        "incomplete",
        "five_calls",
        "fake_attempt",
        "summary_missing",
        "missing_review",
        "bad_visual",
        "missing_QC",
        "changed_QC",
        "stale_consumer",
        "false_predicate",
    ],
)
def test_finite_rejections(case, fault):
    if fault == "incomplete":
        case.program["complete"] = False
    if fault == "five_calls":
        case.program["calls_started"] = 5
    if fault == "fake_attempt":
        case.program["attempt_records"].append(case.program["accepted_records"][2])
    if fault == "summary_missing":
        case.program["files"].pop(str(T.summary_path(case.root, b.G).relative_to(case.root)))
    if fault == "missing_review":
        case.manifest["case_reviews"].pop("g-5")
    if fault in {"bad_visual", "false_predicate"}:
        review = case.reviews["g-3"]
        if fault == "bad_visual":
            review["visual"]["P0_P1_issues"] = 1
        else:
            review["diagnostic_comparison"]["cross_run_numeric_tolerance_used"] = True
        item = case.manifest["case_reviews"]["g-3"]
        item["sha256"] = case.put(Path(item["path"]), review)
    if fault == "missing_QC":
        case.manifest["case_reviews"]["g-1"]["files"].pop(
            next(iter(case.manifest["case_reviews"]["g-1"]["files"]))
        )
    if fault == "changed_QC":
        Path(next(iter(case.manifest["case_reviews"]["g-1"]["files"]))).write_bytes(b"late drift")
    if fault == "stale_consumer":
        case.manifest["consumer_review"]["sha256"] = "0" * 64
    with pytest.raises(ValueError):
        b.build(case.root, case.manifest, root_confirmed_six_closed=True)
    assert not (case.root / T.G_REVIEW).exists()


def test_extra_native_mirror_and_basename_collision_rejected(case):
    qc = Path(next(iter(case.manifest["case_reviews"]["g-1"]["files"]))).parent
    native = qc / "forbidden.nc"
    case.manifest["extra_evidence"][str(native)] = case.put(native, b"native mirror forbidden")
    with pytest.raises(ValueError):
        b.build(case.root, case.manifest, root_confirmed_six_closed=True)
    case.manifest["extra_evidence"] = {}
    first, second = qc / "one" / "same.json", qc / "two" / "same.json"
    case.manifest["extra_evidence"] = {
        str(p): case.put(p, {"different": str(p)}) for p in [first, second]
    }
    with pytest.raises(ValueError):
        b.build(case.root, case.manifest, root_confirmed_six_closed=True)
