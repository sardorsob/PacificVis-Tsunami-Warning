def test_G_uses_three_historical_closures_current_sources_and_reserve(case):
    x = b.build(case.root, T.PHASES[0])
    g = x["gate"]
    assert g["inherited_calls"] == {"charged": 3, "started": 3}
    assert (
        g["source_epoch"] == "technical-completion-v2" and g["proof_contract"] == T.PROOF_CONTRACT
    )
    assert len(case.history) == 3 and len(g["jobs"]) == 6 and len(g["dispatch_jobs"]) == 5
    assert g["files"][T.SELF] != case.old[T.SELF] and T.HISTORY_MANIFEST in g["files"]
    assert case.quota[-1]["reserve"] is True and case.quota[-1]["known_files"] == g["files"]
    assert not (case.root / T.authority_path(T.PHASES[0])).exists()


def test_missing_source_review_fails_before_history_or_quota(case):
    (case.root / case.source).unlink()
    with pytest.raises(ValueError):
        b.build(case.root, T.PHASES[0])
    assert not case.history and not case.quota


def test_full_requires_actual_accepted_G_review_and_six_calls(case):
    with pytest.raises(ValueError):
        b.build(case.root, T.PHASES[1])
    case.put(
        T.G_REVIEW,
        dict(
            status="accepted_frozen_full_development_design_checks",
            independent_checker=True,
            files={},
        ),
    )
    x = b.build(case.root, T.PHASES[1])["gate"]
    assert x["inherited_calls"] == {"charged": 6, "started": 6}
    assert x["proof_contract"] == T.FULL_PROOF_CONTRACT and len(x["dispatch_jobs"]) == 3


@pytest.mark.parametrize(
    "fault", ["stale_source", "new_context", "quota", "floor", "outside", "source_status"]
)
def test_no_laundering_or_budget_failure(case, fault):
    source = T.load(case.root, case.source)
    if fault == "stale_source":
        source["files"][T.SELF] = case.old[T.SELF]
    if fault == "new_context":
        source["files"]["context/live.json"] = case.put("context/live.json", b"no body parsing")
    if fault == "source_status":
        source["execution_authority"] = True
    case.put(case.source, source)
    if fault == "quota":
        case.fake.c2.quota = lambda *a, **kw: "exceeded"
    if fault in {"floor", "outside"}:
        name = T.KERNELS[T.PHASES[0]]
        k = T.load(case.root, name)
        if fault == "floor":
            k["parent_minimum_bytes"].pop(next(iter(k["parent_minimum_bytes"])))
        else:
            k["external_reference_files"].pop(next(iter(k["external_reference_files"])))
        case.put(name, k)
    with pytest.raises(ValueError):
        b.build(case.root, T.PHASES[0])
