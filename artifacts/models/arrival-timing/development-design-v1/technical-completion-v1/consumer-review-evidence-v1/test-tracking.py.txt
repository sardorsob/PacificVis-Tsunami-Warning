class Client:
    def __init__(self):
        self.runs = {}
        self.artifacts = {}
        self.factories = 0
        self.failure = None

    def factory(self, tracking_uri):
        self.factories += 1
        self.store = Path(unquote(urlsplit(tracking_uri).path))
        assert tracking_uri.endswith("/.mlflow/mlruns")
        return self

    def get_experiment_by_name(self, name):
        assert name == "arrival-model"
        return SimpleNamespace(
            experiment_id="1",
            lifecycle_stage="active",
            artifact_location=(self.store / "1").as_uri(),
        )

    def search_runs(self, ids, filter_string, run_view_type):
        assert ids == ["1"] and run_view_type == 3
        field, value = filter_string.split(" = ")
        key = field.removeprefix("tags.").strip("`")
        value = value.strip("'")
        return [r for r in self.runs.values() if r.data.tags.get(key) == value]

    def create_run(self, experiment_id, tags):
        rid = str(len(self.runs) + 1)
        self.runs[rid] = SimpleNamespace(
            info=SimpleNamespace(
                run_id=rid,
                experiment_id=experiment_id,
                lifecycle_stage="active",
                status="RUNNING",
                artifact_uri=(self.store / experiment_id / rid / "artifacts").as_uri(),
            ),
            data=SimpleNamespace(params={}, metrics={}, tags=tags.copy()),
        )
        self.artifacts[rid] = {}
        if self.failure == "create":
            self.failure = None
            raise RuntimeError("ambiguous create after success")
        return self.runs[rid]

    def get_run(self, rid):
        return self.runs[rid]

    def list_artifacts(self, rid):
        return [SimpleNamespace(path=n, is_dir=False) for n in self.artifacts[rid]]

    def log_param(self, rid, key, value, **kw):
        self.runs[rid].data.params[key] = str(value)

    def log_metric(self, rid, key, value, **kw):
        self.runs[rid].data.metrics[key] = value

    def log_artifact(self, rid, name):
        self.artifacts[rid][Path(name).name] = Path(name).read_bytes()
        if self.failure == "artifact":
            self.failure = None
            raise RuntimeError("ambiguous artifact after success")

    def download_artifacts(self, rid, name, dest):
        p = Path(dest) / name
        p.write_bytes(self.artifacts[rid][name])
        return str(p)

    def set_terminated(self, rid, status):
        self.runs[rid].info.status = status

@pytest.mark.parametrize(
    "fault", ["missing", "pending", "not_independent", "missing_source", "report_hash"]
)

    report, reader = fixture(tmp_path)
    client = Client()
    p = report / "review-acceptance.json"
    value = json.loads(p.read_text())
    if fault == "missing":
        p.unlink()
    else:
        if fault == "pending":
            value["status"] = "pending"
        if fault == "not_independent":
            value["independent_checker"] = False

            value["files"].pop("producer/check0/result.json")

        p.write_bytes(m.payload(value))

        m.register(tmp_path, report, reader=reader, execute=True, client_factory=client.factory)
    assert not client.factories and not (tmp_path / m.OUTPUT).exists()

@pytest.mark.parametrize(
    "fault", ["duplicate", "tag", "metric", "artifact", "deleted", "wrong_failed_status"]
)
def test_existing_conflicts_never_duplicate_or_recolor(tmp_path, fault):
    report, reader = fixture(tmp_path)
    client = Client()
    result = m.register(
        tmp_path, report, reader=reader, execute=True, client_factory=client.factory
    )
    rid = result["children"][0]["run_id"]
    run = client.runs[rid]
    if fault == "duplicate":
        client.runs["foreign"] = copy.deepcopy(run)
    if fault == "tag":
        run.data.tags["project_root"] = "another project"
    if fault == "metric":
        run.data.metrics["diagnostics.rhat"] = 9
    if fault == "artifact":
        client.artifacts[rid]["unknown.json"] = b"unknown"
    if fault == "deleted":
        run.info.lifecycle_stage = "deleted"
    if fault == "wrong_failed_status":
        run.info.status = "FINISHED"
    count = len(client.runs)
    with pytest.raises(ValueError):
        m.register(tmp_path, report, reader=reader, execute=True, client_factory=client.factory)
    assert len(client.runs) == count and run.info.status == (
        "FINISHED" if fault == "wrong_failed_status" else "FAILED"
    )

@pytest.mark.parametrize("which", ["parent", "child"])

        tmp_path, report, reader=reader, execute=True, client_factory=client.factory
    )

        m.register(tmp_path, report, reader=reader, execute=True, client_factory=client.factory)

        "A completed immutable receipt must reject a missing recorded run before creation"

@pytest.mark.parametrize(
    "fault",
    [
        "omitted",
        "omitted_all",
        "extra",
        "original_live_state",
        "continuation_live_state",
        "mismatched_hash",
        "non_json",
        "embedded_context",
        "freeze_hash",
        "inherited_live_state",
    ],
)
def test_invalid_legacy_reference_map_stops_before_context_hash_or_client(
    tmp_path, monkeypatch, fault
):
    report, reader = fixture(tmp_path)
    client = Client()
    review_path = report / "review-acceptance.json"
    review = json.loads(review_path.read_text())
    pins = review["files"]
    freeze_path = tmp_path / m.FREEZE
    freeze = json.loads(freeze_path.read_text())
    key = next(iter(freeze["files"]))
    live = "context/runs/arrival-model-fitting-v1/development-state.json"
    if fault == "omitted":
        pins.pop(key)
    elif fault == "omitted_all":
        for name in list(pins):
            if name.startswith("context/"):
                pins.pop(name)
    elif fault == "extra":
        pins["context/unreviewed.json"] = "f" * 64
    elif fault == "original_live_state":
        pins[live] = "f" * 64
    elif fault == "continuation_live_state":
        pins["context/runs/arrival-model-fitting-v1/development-completion-state.json"] = "f" * 64
    elif fault == "mismatched_hash":
        pins[key] = "f" * 64
    elif fault == "embedded_context":
        pins["artifacts/context/unreviewed.json"] = "f" * 64
    elif fault == "freeze_hash":
        pins[m.FREEZE] = "f" * 64
    else:
        destination = key.removesuffix(".json") + ".txt" if fault == "non_json" else live
        freeze["files"][destination] = freeze["files"].pop(key)
        pins[destination] = pins.pop(key)
        freeze_path.write_bytes(m.payload(freeze))
        pins[m.FREEZE] = m.sha(freeze_path)
    review_path.write_bytes(m.payload(review))
    original_sha = m.sha

    def no_context_hash(p):
        assert "context/" not in str(p.relative_to(tmp_path)), "context hash before admission"
        return original_sha(p)

    monkeypatch.setattr(m, "sha", no_context_hash)
    with pytest.raises(ValueError):
        m.register(tmp_path, report, reader=reader, execute=True, client_factory=client.factory)
    assert client.factories == 0 and not (tmp_path / m.OUTPUT).exists()
