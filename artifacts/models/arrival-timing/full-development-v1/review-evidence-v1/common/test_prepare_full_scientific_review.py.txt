def test_no_closed_notice_does_not_touch_program(case):
    with pytest.raises(ValueError):
        b.build(case.root, case.manifest)
    assert not case.calls


def test_three_full_fits_nine_calls_and_complete_portable_plan(case):
    x = b.build(case.root, case.manifest, root_confirmed_three_closed=True)
    r = x["review"]
    assert r["status"] == "accepted_frozen_full_development_fits"
    assert (
        r["development_check_previously_inspected"] is True
        and r["confirmation_outcomes_parsed"] is True
    )
    assert r["reserved_outcomes"] == "sealed" and r["reserved_outcomes_parsed"] is False
    assert len(r["accepted_records"]) == len(r["attempt_records"]) == 3
    assert (
        r["calls_charged"] == r["calls_started"] == 9
        and r["proof_contract"] == T.FULL_PROOF_CONTRACT
    )
    assert r["files"][T.G_REVIEW] == case.program["files"][T.G_REVIEW]
    assert all(p.startswith(b.EVIDENCE + "/") for p in x["copy_plan"])
    assert not (case.root / case.target).exists()


@pytest.mark.parametrize(
    "fault",
    [
        "incomplete",
        "calls8",
        "missingSCI",
        "job",
        "phase",
        "diag",
        "visual",
        "numeric",
        "producer",
        "missing_QC",
        "late_QC",
        "Readerclosure",
    ],
)
def test_refusals(case, fault):
    if fault == "incomplete":
        case.program["complete"] = False
    if fault == "calls8":
        case.program["calls_started"] = 8
    if fault == "missingSCI":
        case.manifest["case_reviews"].pop("full-2")
    if fault == "Readerclosure":
        case.program["files"].pop("reader.py")
    if fault in ["job", "phase", "diag", "visual", "numeric", "producer"]:
        r = case.reviews["full-0"]
        if fault == "job":
            r["job"] = {**r["job"], "order": 99}
        if fault == "phase":
            r["phase"] = "full_design_recovery"
        if fault == "diag":
            r["diagnostics"]["divergences"] = 1
        if fault == "visual":
            r["visual"]["P0_P1_issues"] = 1
        if fault == "numeric":
            r["numerics"]["passed"] = False
        if fault == "producer":
            r["producer_git_sha"] = "b" * 40
        item = case.manifest["case_reviews"]["full-0"]
        item["sha256"] = case.put(Path(item["path"]), r)
    if fault == "missing_QC":
        case.manifest["case_reviews"]["full-0"]["files"].pop(
            next(iter(case.manifest["case_reviews"]["full-0"]["files"]))
        )
    if fault == "late_QC":
        Path(next(iter(case.manifest["case_reviews"]["full-0"]["files"]))).write_bytes(b"late")
    with pytest.raises(ValueError):
        b.build(case.root, case.manifest, root_confirmed_three_closed=True)
    assert not (case.root / case.target).exists()
