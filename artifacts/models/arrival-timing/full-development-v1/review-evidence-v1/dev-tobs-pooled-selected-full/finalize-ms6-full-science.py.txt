def require(ok, message):
    if not ok:
        raise ValueError(message)


def sha(path):
    with Path(path).open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def read(path):
    return json.loads(Path(path).read_text())


def fresh(root, files):
    require(type(files) is dict and files, "nonempty exact byte closure")
    for name, digest in files.items():
        require(
            isinstance(name, str)
            and not Path(name).is_absolute()
            and ".." not in Path(name).parts
            and name not in PROTECTED,
            "repo-relative scientific/producer closure without live private state",
        )
        path = root / name
        require(
            path.resolve() == path and path.is_file() and sha(path) == digest,
            "fresh exact scientific source/producer bytes",
        )


def finalize(
    root,
    output,
    *,
    notice_path,
    process_path,
    runtime_path,
    chart_path,
    chart_code_path,
    visual_path,
):
    root, output = Path(root).resolve(), Path(output).absolute()
    require(sha(AUDIT) == AUDIT_SHA, "frozen accepted audit entry")
    spec = importlib.util.spec_from_file_location("full_receipt_closed_guard", AUDIT)
    assert spec and spec.loader
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    notice, process = read(notice_path), read(process_path)
    jid = process.get("job_id")
    module.closed_guard(notice, process, PHASE, jid)
    require(
        output == TMP_ROOT / ("ms6-accepted-science-" + jid) and output.resolve() == output,
        "literal actual TEMP case folder",
    )
    destination = output / "independent-scientific-review.json"
    require(not destination.exists(), "new exclusive independent receipt")
    # No audit body, native/table body, model or scientific import precedes CLOSED proof.
    audit_path = output / "scientific-check.json"
    audit = read(audit_path)
    require(
        audit.get("status") == "independent_numerics_passed_visual_review_pending"
        and audit.get("phase") == PHASE
        and audit.get("job") == process.get("job")
        and audit.get("generation") is None
        and audit.get("complete_artifact_replay_repeated") is False
        and audit.get("numerical_oracle_source_repeated") is False
        and type(audit.get("new_sampler_calls_in_QA")) is int
        and audit["new_sampler_calls_in_QA"] == 0,
        "closed original full fit independent arithmetic only; no simulation truth",
    )
    require(
        audit.get("density", {}).get("passed") is True
        and audit.get("summary", {}).get("passed") is True,
        "independent density/gradient and seeded summaries must pass",
    )
    diagnostics = audit.get("diagnostics", {})
    require(
        diagnostics.get("screen_passed") is True
        and diagnostics.get("finite_diagnostics") is True
        and type(diagnostics.get("divergences")) is int
        and diagnostics["divergences"] == 0
        and type(diagnostics.get("maxdepth_reached")) is int
        and diagnostics["maxdepth_reached"] == 0,
        "original full-fit native diagnostic screen",
    )
    for name, digest in process["files"].items():
        require(audit["files"].get(name) == digest, "entire immutable producer/P closure")
    fresh(root, audit["files"])
    folder = root / "artifacts/models/arrival-timing/full-development-v1/jobs" / jid
    raw = root / "data/processed/models/arrival-development-v1" / (jid + ".nc")

        isinstance(git_sha, str)
        and len(git_sha) == 40
        and all(c in "0123456789abcdef" for c in git_sha)

    )
    require(
        visual.get("P0_P1_issues") == 0
        and type(visual.get("P0_P1_issues")) is int
        and visual.get("chart_fails") == 0
        and type(visual.get("chart_fails")) is int
        and visual.get("native_figures_reopened") is True
        and visual.get("six_hundred_pixel_previews_reopened") is True,
