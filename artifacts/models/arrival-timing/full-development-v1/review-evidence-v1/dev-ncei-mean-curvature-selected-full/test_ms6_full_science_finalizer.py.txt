def module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    assert spec and spec.loader
    value = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(value)
    return value

def test_exact_full_schema_no_truth_or_scientific_import_and_exclusive(tmp_path, monkeypatch):
    root, output, audit, args = fixture(tmp_path, monkeypatch)
    result = q.finalize(root, output, **args)
    assert result["status"] == "accepted_closed_full_development_scientific_checks"
    assert (
        result["generation"] is None
        and result["truth_validation"] == "not_applicable_full_data_fit"
    )
    assert result["numerics"] == audit["density"] and result["summary"] == audit["summary"]
    assert result["source_files"] == audit["files"]
    assert result["visual"]["native_figures_reopened"] is True
    assert (
        result["new_sampler_calls_in_QA"] == 0
        and result["private_state_bodies_parsed_or_copied"] is False
    )
    with pytest.raises(ValueError, match="exclusive"):
        q.finalize(root, output, **args)


@pytest.mark.parametrize(
    "fault",
    [
        "closed",
        "G6",
        "diagnostic",
        "boolcost",
        "generation",
        "density",
        "divergence",
        "depth",
        "visual",
        "producer",
        "source",
        "private_pin",
    ],
)
def test_fail_closed_and_no_receipt_on_corruption(tmp_path, monkeypatch, fault):
    root, output, audit, args = fixture(tmp_path, monkeypatch)
    if fault in ["closed", "G6"]:
        name = "notice_path"
        v = q.read(args[name])
        v["closed" if fault == "closed" else "accepted_G6_review"] = False
    elif fault in ["diagnostic", "boolcost"]:
        name = "process_path"
        v = q.read(args[name])
        v["diagnostic_scope_only" if fault == "diagnostic" else "cost"] = True
    elif fault == "visual":
        name = "visual_path"
        v = q.read(args[name])
        v["six_hundred_pixel_previews_reopened"] = False
    elif fault == "producer":
        name = "runtime_path"
        v = q.read(args[name])
        v["producer_git_sha"] = "2" * 40
    else:
        name = None
        v = None
        if fault == "generation":
            audit["generation"] = {"not_full_fit": True}
        elif fault == "density":
            audit["density"]["passed"] = False
        elif fault == "divergence":
            audit["diagnostics"]["divergences"] = True
        elif fault == "depth":
            audit["diagnostics"]["maxdepth_reached"] = 1
        elif fault == "private_pin":
            audit["files"].update(q.PROTECTED)
        else:
            (root / next(iter(audit["files"]))).write_text("changed retained bytes")
    if name:
        args[name].write_text(json.dumps(v))
    (output / "scientific-check.json").write_text(json.dumps(audit))
    with pytest.raises(ValueError):
        q.finalize(root, output, **args)
    assert not (output / "independent-scientific-review.json").exists()
