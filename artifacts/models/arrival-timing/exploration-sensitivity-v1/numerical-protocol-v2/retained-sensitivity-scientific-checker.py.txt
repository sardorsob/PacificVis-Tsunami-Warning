def selected(state,start,stop):
    if not (35<=start<stop<=55 and stop-start<=12):
        raise ValueError('one bounded accepted slice,35<=start<stop<=55,count<=12')
    entry=state.get('completion_v1') or {}; registry=entry.get('accepted_remaining',[])
    if len(registry)<stop-35:raise ValueError('requested case is not yet accepted; no raw body may be opened')
    return registry[start-35:stop-35]

def weights(distance):
    return np.column_stack([np.interp(np.log(distance),np.log([500,2000,5000]),np.eye(3)[j]) for j in range(3)])

def design(distance,curved):
    columns=[np.ones(len(distance)),np.log(distance/1000)]
    if curved:columns.append(np.log((distance+500)/(1.5*distance)))
    return np.column_stack(columns)

def independent_prior(q,prior,curved):
    ncoef=3 if curved else 2; coeff=q[:ncoef];positive=np.exp(q[ncoef:]);scales=np.array([prior[k] for k in ['alpha_sd','beta_sd']+(['gamma_sd'] if curved else [])])
    pscales=np.array([prior['tau_sd']]+[prior['sigma_sd']]*3)
    value=float(norm.logpdf(coeff,scale=scales).sum()+norm.logpdf(positive,scale=pscales).sum()+4*np.log(2)+q[ncoef:].sum())
    gradient=np.r_[-coeff/scales**2,1-positive**2/pscales**2]
    return value,gradient

def independent_joint(q,distance,response,event,prior,curved):
    X=design(distance,curved);w=weights(distance);ncoef=X.shape[1];tau=np.exp(q[ncoef]);anchors=np.exp(q[ncoef+1:]);sigma=w@anchors
    residual=response-X@q[:ncoef];value,gprior=independent_prior(q,prior,curved);g=np.zeros_like(q)
    for name in sorted(set(event)):
        ix=np.flatnonzero(event==name);sd=sigma[ix];C=np.diag(sd**2)+tau**2*np.ones((len(ix),len(ix)));factor=cho_factor(C,lower=True);z=cho_solve(factor,residual[ix]);inv=cho_solve(factor,np.eye(len(ix)))
        value+=-.5*(len(ix)*np.log(2*np.pi)+2*np.log(np.diag(factor[0])).sum()+residual[ix]@z)
        g[:ncoef]+=X[ix].T@z
        g[ncoef]+=tau**2*(z.sum()**2-inv.sum())
        g[ncoef+1:]+=anchors*(w[ix].T@(sd*(z*z-np.diag(inv))))
    return float(value),g+gprior

def predictive_replay(job,post,observed,summary):
    hyper={k:v.values for k,v in post.data_vars.items()};n=post.sizes['chain']*post.sizes['draw'];seeds=np.random.SeedSequence(job['seed']+1).generate_state(4);take=np.sort(np.random.default_rng(seeds[0]).choice(n,800,replace=False));np.testing.assert_array_equal(take,summary['posterior_flat_indices']);np.testing.assert_array_equal(seeds,summary['stream_seeds'])
    d=observed.distance_km.values;y=observed.response.values;events=np.unique(observed.event_id.values)
    def components(pop,dist):
        mu=pop['alpha'][...,None]+pop['beta'][...,None]*np.log(dist/1000)
        if 'gamma' in pop:mu=mu+pop['gamma'][...,None]*np.log((dist+500)/(1.5*dist))
        sd=np.einsum('...j,ij->...i',pop['sigma_anchor'],weights(dist));return mu,sd
    mu,sd=components(hyper,d);noise=np.random.default_rng(seeds[1]).normal(size=(4,2000,len(events)));effects=np.empty_like(noise)
    for j,event in enumerate(events):
        ix=observed.event_id.values==event;precision=1/sd[:,:,ix]**2;total=precision.sum(-1);variance=hyper['tau_event']**2/(1+hyper['tau_event']**2*total);mean=variance*np.sum(precision*(y[ix]-mu[:,:,ix]),axis=-1);effects[:,:,j]=mean+np.sqrt(variance)*noise[:,:,j]
    pop={k:v.reshape((n,*v.shape[2:]))[take] for k,v in hyper.items()};mu,sd=components(pop,d);conditional=mu+effects.reshape(n,len(events))[take][:,np.searchsorted(events,observed.event_id.values)]+sd*np.random.default_rng(seeds[2]).normal(size=mu.shape);native=conditional+np.log(observed.reference_minutes.values)
    assert np.isfinite(conditional).all() and np.isfinite(native).all()

    grid=np.array([20.,50.,100.,500.,1000.,2000.,5000.,10000.,20000.]);mu,sd=components(pop,grid);rng=np.random.default_rng(seeds[3]);new=mu+pop['tau_event'][:,None]*rng.normal(size=(800,1))+sd*rng.normal(size=mu.shape);new+=np.log(grid/(np.sqrt(9.80665*4000)*60/1000));assert np.isfinite(new).all()

    for row in summary['derived']:
        dist=row['distance_km'];v=components({k:v.reshape((n,*v.shape[2:])) for k,v in hyper.items()},np.array([dist]))[1][:,0] if row['quantity']=='residual_sd_modeled_response' else 1+hyper['beta'].ravel()-hyper.get('gamma',np.zeros((4,2000))).ravel()*500/(dist+500);np.testing.assert_allclose([row[k] for k in ['q05','q50','q95']],np.quantile(v,[.05,.5,.95]),rtol=1e-12,atol=1e-12)
    misses=[]
    for row in summary['conditional_ppc']:
        label=row['stratum'];mask={'all':np.ones(len(d),bool),'under-500km':d<500,'500-to-2000km':(d>=500)&(d<2000),'2000-to-5000km':(d>=2000)&(d<5000),'5000km-and-above':d>=5000}[label];data=conditional if row['scale']=='modeled-response' else native;obs=y if row['scale']=='modeled-response' else np.log(observed.native_minutes.values);q=np.quantile(np.quantile(data[:,mask],row['quantile'],axis=1),[.05,.5,.95]);oq=np.quantile(obs[mask],row['quantile'])
        if row['scale']=='native-minutes':q=np.exp(q);oq=np.exp(oq)

def independent_mask(lane,metadata,scenario):
    ids=np.array(lane.record_ids);years=np.array([metadata.event_years[e] for e in lane.event_ids])
    rules={
        'coastal-only':metadata.coastal,
        'era-pre2007':years<2007,
        'era-2007-2013':(years>=2007)&(years<=2013),
        'era-2014plus':years>=2014,
        'publisher-distance':np.ones(len(ids),bool),
        'omit-triage':~np.isin(ids,metadata.triage_record_ids),
        'component-assessed':np.isin(ids,metadata.component_assessed_record_ids),
        'omit-native-below10':lane.y>=10,
        'omit-lata-only':ids!=metadata.lata_record_id,
        'omit-tohoku-and-chile':~np.isin(lane.sequences,[metadata.influence_sequences['tohoku'],metadata.influence_sequences['chile2010']]),
        'omit-definition':~np.isin(ids,metadata.definition_record_ids),
    }
    return np.flatnonzero(lane.included&rules[scenario])
