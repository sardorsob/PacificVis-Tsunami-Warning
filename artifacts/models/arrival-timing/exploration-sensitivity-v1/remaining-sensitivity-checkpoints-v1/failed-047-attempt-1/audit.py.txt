def scalar(value):
    value=np.asarray(value).item()
    if isinstance(value,(float,np.floating)) and not np.isfinite(value):return None
    return value

def dense(q,d,y,event,prior):
    X=old.design(d,True);sigma=old.weights(d)@np.exp(q[4:]);unique=np.unique(event);Z=(event[:,None]==unique[None,:]).astype(float)
    C=np.diag(sigma*sigma)+np.exp(2*q[3])*(Z@Z.T);factor=cho_factor(C,lower=True);r=y-X@q[:3]
    return float(-.5*(len(y)*np.log(2*np.pi)+2*np.log(np.diag(factor[0])).sum()+r@cho_solve(factor,r))+old.independent_prior(q,prior,True)[0])
