def select_closed(state,start,stop):
 if not (48<=start<stop<=55 and stop-start<=4):raise ValueError('explicit original48..54 closed slice, count<=4 required')
 entry=state.get('finish_runs',{}).get('finish-sensitivities-v1',{})
 if entry.get('current') is not None or entry.get('status') not in {'checkpoint','computed_pending_review'}:raise ValueError('only typed, idle successful finish prefixes may be reviewed')
 outcome=entry.get('case47_outcome')
 if outcome not in {'accepted_repair','exhausted_unresolved'} or not entry.get('case47_review_sha256'):raise ValueError('typed independent case47 disposition required before tail SCI')
 accepted=entry.get('accepted',[]);orders=[x.get('order') for x in accepted]
 if len(orders)!=len(set(orders)):raise ValueError('duplicate original order in finish accepted registry')
 by_order={d['order']:d for d in accepted}
 if (outcome=='accepted_repair') != (47 in by_order):raise ValueError('case47 outcome/accepted registry mismatch')
 if any(i not in by_order for i in range(start,stop)):raise ValueError('incomplete tail raw must not be opened')
 if state.get('calls_charged')!=state.get('calls_started') or state['calls_charged']!=50+len(entry.get('attempts',[])):raise ValueError('true finish call conservation required')
 return entry,[by_order[i] for i in range(start,stop)]
