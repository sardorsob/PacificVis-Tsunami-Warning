def dense(q, d, y, event, prior):
    X = old.design(d, True)
    sigma = old.weights(d) @ np.exp(q[4:])
    Z = (event[:, None] == np.unique(event)[None, :]).astype(float)
    C = np.diag(sigma * sigma) + np.exp(2 * q[3]) * (Z @ Z.T)
    factor = cho_factor(C, lower=True)
    residual = y - X @ q[:3]
    return float(-.5 * (len(y) * np.log(2 * np.pi) + 2 * np.log(np.diag(factor[0])).sum()
                        + residual @ cho_solve(factor, residual))
                 + old.independent_prior(q, prior, True)[0])


def closed(state, admission):
    entry = state.get('finish_runs', {}).get('finish-sensitivities-v1', {})
    assert state['calls_charged'] == state['calls_started'] == 51
    assert len(entry.get('attempts', [])) == 1
    assert entry['attempts'][0]['execution_job'] == admission['repair_job']
    good = entry['attempts'][0]['outcome'] == 'accepted'
    assert entry['status'] == ('awaiting_case47_review' if good else 'needs_review')
    assert entry['current'] == (None if good else admission['repair_job'])
    assert entry['next_job'] == (1 if good else 0)
    assert len(entry['accepted']) == int(good)
    assert not entry.get('case47_review_sha256')
    return entry, good

    physical = m.physical_files(root, folder, raw)

        m.new_provenance(root, admission, accepted)
        # This replay is unconditional: accepted metadata does not replace native evidence.

        sources = m.p.merge(sources, proof['files'], {accepted['receipt']: accepted['receipt_sha256']})
    else:
        accepted = None
        assert not (folder / 'summary.json').exists()
        assert not any((folder / (g.replace('_', '-') + '.csv')).exists() for g in worker.TABLES)
    m.c._unchanged(root, sources)

    assert frozen == original['contract']
    meta = source_io.load_sensitivity_inputs(root, lanes, original['files'])
    jobs = fit.sensitivity_jobs(build_sensitivity_design(lanes, meta))
    assert jobs == admission['worker_admission']['original_jobs'] and job == jobs[47]
    original_lane = lanes[job['target']]
    lane = case_lane(original_lane, job, meta)
    ids = case_indices(original_lane, job)

    assert len(ids) == 150 and job['kind'] == 'mean-curvature' and job['prior_variant'] == 'base'
    assert lane.record_ids == original_lane.record_ids
    np.testing.assert_array_equal(lane.y, original_lane.y)

    assert job['omitted_record_ids'] == [rid for rid, keep in zip(lane.record_ids, original_lane.included, strict=True) if keep and rid not in set(job['training_record_ids'])]
    assert job['excluded_record_ids'] == [rid for rid, keep in zip(lane.record_ids, original_lane.included, strict=True) if not keep]
    assert len(job['excluded_record_ids']) == 6
    np.testing.assert_allclose(lane.reference_minutes, lane.distance_km / (np.sqrt(9.80665 * 4000) * 60 / 1000), rtol=1e-14, atol=1e-14)
    np.testing.assert_allclose(lane.response, np.log(lane.y) - np.log(lane.reference_minutes), rtol=1e-14, atol=1e-14)
    base, wider = fit._case_priors(root, job)
    assert wider is None
    inputs, membership = worker._metadata(execution, {**admission, 'files': admission['initial_files'], 'gate_path': m.GATE}, lane, ids, frozen, base)

    assert read(str((folder / 'membership.json').relative_to(root))) == membership

    old_inputs = read(str((old_folder / 'inputs.json').relative_to(root)))

    assert membership == read(str((old_folder / 'membership.json').relative_to(root)))
    result = read(str((folder / 'result.json').relative_to(root)))
    assert result['job'] == execution and result['original_job'] == job
    assert result['code_and_config'] == admission['initial_files']
    assert result['posterior_sha256'] == sha256(raw)
    assert result['confirmation_outcomes_parsed'] is False and result['reserved_outcomes'] == 'sealed'
    assert result['scoring_or_forecast_imputation'] is False
    process = read(str(folder.with_name(folder.name + '.supervisor.json').relative_to(root)))
    assert process['returncode'] == (0 if healthy else 2) and process['reason'] == 'worker_exit'
    assert process['charged_slots'] == process['start_markers'] == 1 and not process['monitor_failures']
    assert process['numerical_gate_sha256'] == process['execution_authority_gate_sha256'] == GATE_SHA

    for name in names:
